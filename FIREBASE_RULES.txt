rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions for authorization
    function isAuthenticated() {
      return request.auth != null;
    }

    function isAdmin() {
      return isAuthenticated() && (
        request.auth.token.email == 'aneeb458@gmail.com' ||
        exists(/databases/$(database)/documents/admins/$(request.auth.uid))
      );
    }

    // Menu items: Publicly viewable, editable only by admins
    // Note: Stock deduction is executed safely server-side by Cloud Functions
    match /items/{itemId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Categories: Publicly viewable, editable only by admins
    match /categories/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Sliders & promotional banners: Publicly viewable, editable only by admins
    match /sliders/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Settings (operating hours, store info): Publicly viewable, editable only by admins
    match /settings/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Admins registry: readable by authenticated users (to verify permissions), writable only by admins
    match /admins/{adminId} {
      allow read: if isAuthenticated();
      allow write: if isAdmin();
    }

    // Orders:
    // - Anyone can create/place an order
    // - Read access restricted to: admins, order owner (customer), or assigned rider
    // - Update access restricted to: admins or assigned rider (status progression)
    // - Delete access restricted to: admins only
    match /orders/{orderId} {
      allow create: if true;
      allow read: if isAdmin() || (
        isAuthenticated() && (
          (resource.data.userId != null && resource.data.userId == request.auth.uid) ||
          (resource.data.assignedRiderId != null && resource.data.assignedRiderId == request.auth.uid)
        )
      );
      allow update: if isAdmin() || (
        isAuthenticated() && resource.data.assignedRiderId != null && resource.data.assignedRiderId == request.auth.uid
      );
      allow delete: if isAdmin();
    }

    // Riders collection:
    // - Newly registered riders can create and update their own profile
    // - Admins have full access; riders can read only their own profile
    match /riders/{riderId} {
      allow create: if isAuthenticated() && request.auth.uid == riderId;
      allow read: if isAdmin() || (isAuthenticated() && request.auth.uid == riderId);
      allow update: if isAdmin() || (isAuthenticated() && request.auth.uid == riderId);
      allow delete: if isAdmin();
    }

    // Rider signup codes:
    // - Unused codes can be read during registration validation
    // - Only admins can generate or delete codes
    // - Newly registered riders can mark a code as used during registration
    match /riderSignupCodes/{codeId} {
      allow read: if (resource != null && resource.data.used == false) || isAdmin();
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        isAuthenticated() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['used', 'usedAt', 'usedBy']) &&
        request.resource.data.used == true
      );
    }

    // Inventory & Raw Materials: STRICTLY admin-only
    match /inventory/{document=**} {
      allow read, write: if isAdmin();
    }

    // Financial Expenses: STRICTLY admin-only
    match /expenses/{document=**} {
      allow read, write: if isAdmin();
    }

    // Default catch-all: deny everything else
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

// ============================================
// FIREBASE STORAGE RULES
// ============================================
service firebase.storage {
  match /b/{bucket}/o {
    // Anyone can read public restaurant assets (menu images, banners)
    match /{allPaths=**} {
      allow read: if true;
    }

    // Admins only can upload images (Max 5MB, images only)
    match /menu-images/{imageId} {
      allow write: if request.auth != null
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    match /slider-images/{imageId} {
      allow write: if request.auth != null
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    // Other uploads by authenticated staff
    match /{allPaths=**} {
      allow write: if request.auth != null
                   && request.resource.size < 10 * 1024 * 1024;
    }
  }
}
